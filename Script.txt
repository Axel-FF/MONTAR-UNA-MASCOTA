local function ApplyMove()
	local h = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")

		h.WalkSpeed = S.SpeedVal

local IGNORAR = {"horn","brow","fur","button","eye","mouth","leg","arm","hand","foot","head","torso","mesh","fp_"}
local function EsPieza(n)
	n = n:lower()
	for _, p in ipairs(IGNORAR) do if n:find(p) then return true end end
	return false
end

local function GetHuevos()
	local lista, vistos = {}, {}
	for _, obj in pairs(Workspace:GetDescendants()) do
		if obj:IsA("Model") then
			local nom = obj.Name:lower()
			if (nom:find("egg") or nom:find("huevo")) and not EsPieza(obj.Name) and not vistos[obj] then
				local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
				if part and part.Transparency < 1 then
					vistos[obj] = true
					table.insert(lista, {obj=obj, part=part, pos=part.Position})
				end
			end
		end
	end
	return lista
end

local function ClearESP()
	for _, d in pairs(ESPObjects) do
		pcall(function() if d.hl then d.hl:Destroy() end if d.bill then d.bill:Destroy() end end)
	end
	table.clear(ESPObjects)
end

local function MakeESP(obj, part)
	if ESPObjects[obj] then return end
	local d = {}
	local hl = Instance.new("Highlight")
	hl.Adornee = obj
	hl.FillColor = Color3.fromRGB(255, 255, 255)
	hl.OutlineColor = Color3.fromRGB(255, 215, 0)
	hl.FillTransparency = 0.4
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	hl.Parent = obj
	d.hl = hl
	local bill = Instance.new("BillboardGui")
	bill.Size = UDim2.new(0, 140, 0, 36)
	bill.AlwaysOnTop = true
	bill.StudsOffset = Vector3.new(0, 3, 0)
	bill.Adornee = part
	bill.Parent = part
	local nameL = Instance.new("TextLabel")
	nameL.Size = UDim2.new(1, 0, 0, 18)
	nameL.BackgroundTransparency = 1
	nameL.Text = "🥚 " .. obj.Name
	nameL.Font = Enum.Font.GothamBold
	nameL.TextSize = 12
	nameL.TextColor3 = Color3.fromRGB(255, 220, 0)
	nameL.TextStrokeTransparency = 0
	nameL.Parent = bill
	local distL = Instance.new("TextLabel")
	distL.Size = UDim2.new(1, 0, 0, 14)
	distL.Position = UDim2.new(0, 0, 0, 18)
	distL.BackgroundTransparency = 1
	distL.Text = "0m"
	distL.Font = Enum.Font.Gotham
	distL.TextSize = 11
	distL.TextColor3 = Color3.new(1, 1, 1)
	distL.TextStrokeTransparency = 0
	distL.Parent = bill
	d.bill, d.distL = bill, distL
	ESPObjects[obj] = d
end

local last = 0
RunService.RenderStepped:Connect(function()
	if not S.ESP or tick() - last < 0.25 then return end
	last = tick()
	local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local activos = {}
	for _, h in ipairs(GetHuevos()) do
		activos[h.obj] = true
		if not ESPObjects[h.obj] then MakeESP(h.obj, h.part) end
		local d = ESPObjects[h.obj]
		if d and d.distL then
			d.distL.Text = math.floor((h.pos - root.Position).Magnitude) .. "m"
		end
	end
	for obj, d in pairs(ESPObjects) do
		if not activos[obj] or not obj.Parent then
			pcall(function() if d.hl then d.hl:Destroy() end if d.bill then d.bill:Destroy() end end)
			ESPObjects[obj] = nil
		end
	end
end)

local function MakeSlider(parent, y, name, minV, maxV, startV, cb)
	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, 0, 0, 18)
	lbl.Position = UDim2.new(0, 0, 0, y)
	lbl.BackgroundTransparency = 1
	lbl.Text = name .. ": " .. startV
	lbl.Font = Enum.Font.GothamBold
	lbl.TextSize = 13
	lbl.TextColor3 = Color3.fromRGB(255, 215, 80)
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Parent = parent

	local track = Instance.new("Frame")
	track.Size = UDim2.new(1, 0, 0, 12)
	track.Position = UDim2.new(0, 0, 0, y + 26)
	track.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
	track.Parent = parent
	Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((startV - minV) / (maxV - minV), 0, 1, 0)
	fill.BackgroundColor3 = Color3.fromRGB(212, 175, 55)
	fill.Parent = track
	Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, 18, 0, 18)
	knob.AnchorPoint = Vector2.new(0.5, 0.5)
	knob.Position = UDim2.new((startV - minV) / (maxV - minV), 0, 0.5, 0)
	knob.BackgroundColor3 = Color3.new(1, 1, 1)
	knob.Parent = track
	Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

	local drag = false
	local function upd(x)
		local a, w = track.AbsolutePosition.X, track.AbsoluteSize.X
		if w < 1 then return end
		local p = math.clamp((x - a) / w, 0, 1)
		local val = math.floor(minV + p * (maxV - minV) + 0.5)
		fill.Size = UDim2.new(p, 0, 1, 0)
		knob.Position = UDim2.new(p, 0, 0.5, 0)
		lbl.Text = name .. ": " .. val
		cb(val)
	end
	track.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			drag = true
			upd(i.Position.X)
		end
	end)
	UIS.InputChanged:Connect(function(i)
		if drag and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			upd(i.Position.X)
		end
	end)
	UIS.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			drag = false
		end
	end)
	return lbl
end
