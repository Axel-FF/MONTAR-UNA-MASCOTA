-- ===== FPS BOOST =====
local function ActivarFPS()
	pcall(function()
		settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
	end)
	Lighting.GlobalShadows = false
	Lighting.FogEnd = 9e9
	Lighting.Brightness = 1
	Lighting.ClockTime = 14
	pcall(function()
		Lighting.EnvironmentDiffuseScale = 0
		Lighting.EnvironmentSpecularScale = 0
	end)
	for _, fx in pairs(Lighting:GetChildren()) do
		if fx:IsA("BloomEffect") or fx:IsA("SunRaysEffect")
			or fx:IsA("BlurEffect") or fx:IsA("ColorCorrectionEffect")
			or fx:IsA("DepthOfFieldEffect") then
			pcall(function() fx.Enabled = false end)
		end
	end
	for _, v in pairs(Workspace:GetDescendants()) do
		pcall(function()
			if v:IsA("BasePart") then
				v.CastShadow = false
			elseif v:IsA("ParticleEmitter") or v:IsA("Trail")
				or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
				v.Enabled = false
			end
		end)
	end
	local Terrain = Workspace:FindFirstChildOfClass("Terrain")
	if Terrain then
		Terrain.WaterWaveSize = 0
		Terrain.WaterWaveSpeed = 0
		Terrain.WaterReflectance = 0
	end
end
ActivarFPS()

-- ===== ESP + CHAMS =====
local IGNORAR = {
	"horn", "brow", "nostril", "fur", "button", "eye", "mouth",
	"leg", "arm", "hand", "foot", "head", "torso",
	"mesh", "attachment", "weld", "fp_", "left", "right"
}

local function EsPieza(n)
	n = n:lower()
	for _, p in ipairs(IGNORAR) do
		if n:find(p) then return true end
	end
	return false
end

local function GetHuevos()
	local lista, vistos = {}, {}
	for _, obj in pairs(Workspace:GetDescendants()) do
		if obj:IsA("Model") then
			local nom = obj.Name:lower()
			if (nom:find("egg") or nom:find("huevo")) and not EsPieza(obj.Name) and not vistos[obj] then
				local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
				if part and part.Transparency < 1 then
					vistos[obj] = true
					table.insert(lista, {
						obj = obj,
						part = part,
						nombre = obj.Name,
						pos = part.Position
					})
				end
			end
		end
	end
	return lista
end

local function ClearESP()
	for _, d in pairs(ESPObjects) do
		pcall(function()
			if d.hl then d.hl:Destroy() end
			if d.bill then d.bill:Destroy() end
		end)
	end
	table.clear(ESPObjects)
end

local function MakeESP(obj, part)
	if ESPObjects[obj] then return end
	if obj:FindFirstChild("NinjaHL") then return end

	local d = {}
	local hl = Instance.new("Highlight")
	hl.Name = "NinjaHL"
	hl.Adornee = obj
	hl.FillColor = Color3.fromRGB(255, 255, 255)
	hl.OutlineColor = Color3.fromRGB(255, 215, 0)
	hl.FillTransparency = 0.4
	hl.OutlineTransparency = 0
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	hl.Enabled = true
	hl.Parent = obj
	d.hl = hl

	local bill = Instance.new("BillboardGui")
	bill.Name = "NinjaBill"
	bill.Size = UDim2.new(0, 150, 0, 40)
	bill.AlwaysOnTop = true
	bill.StudsOffset = Vector3.new(0, 3, 0)
	bill.Adornee = part
	bill.Parent = part

	local nameL = Instance.new("TextLabel")
	nameL.Size = UDim2.new(1, 0, 0, 20)
	nameL.BackgroundTransparency = 1
	nameL.Text = "🥚 " .. obj.Name
	nameL.Font = Enum.Font.GothamBold
	nameL.TextSize = 13
	nameL.TextColor3 = Color3.fromRGB(255, 220, 0)
	nameL.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	nameL.TextStrokeTransparency = 0
	nameL.Parent = bill

	local distL = Instance.new("TextLabel")
	distL.Size = UDim2.new(1, 0, 0, 16)
	distL.Position = UDim2.new(0, 0, 0, 20)
	distL.BackgroundTransparency = 1
	distL.Text = "0m"
	distL.Font = Enum.Font.Gotham
	distL.TextSize = 11
	distL.TextColor3 = Color3.fromRGB(255, 255, 255)
	distL.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	distL.TextStrokeTransparency = 0
	distL.Parent = bill

	d.bill = bill
	d.distL = distL
	ESPObjects[obj] = d
end

local lastESP = 0
RunService.RenderStepped:Connect(function()
	if not S.ESP then return end
	if tick() - lastESP < 0.2 then return end
	lastESP = tick()

	local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not root then return end

	local activos = {}
	for _, h in ipairs(GetHuevos()) do
		activos[h.obj] = true
		if not ESPObjects[h.obj] then
			MakeESP(h.obj, h.part)
		end
		local d = ESPObjects[h.obj]
		if d then
			if d.hl and d.hl.Parent then
				d.hl.Enabled = true
				d.hl.Adornee = h.obj
			end
			if d.distL then
				d.distL.Text = math.floor((h.pos - root.Position).Magnitude) .. "m"
			end
		end
	end

	for obj, d in pairs(ESPObjects) do
		if not activos[obj] or not obj.Parent then
			pcall(function()
				if d.hl then d.hl:Destroy() end
				if d.bill then d.bill:Destroy() end
			end)
			ESPObjects[obj] = nil
		end
	end
end)
