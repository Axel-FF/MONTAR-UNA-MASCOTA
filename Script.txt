-- ✅ RESTAURAR COLISIONES
local function RestaurarColisiones()
	if not LP.Character then return end
	for _, parte in pairs(LP.Character:GetDescendants()) do
		if parte:IsA("BasePart") then
			parte.CanCollide = true
		end
	end
end

-- ✅ TP INDIVIDUAL
local function IrTP(indice)
	if not LP.Character then return end
	local HRP = LP.Character:FindFirstChild("HumanoidRootPart")
	if not HRP then return end
	local rancho = RANCHOS[indice]
	if not rancho then return end
	HRP.CFrame = CFrame.new(rancho.pos)
	Estado.Text = "✅ TP → " .. rancho.nombre
end

-- ✅ AUTO-FARM
local function IrFarm(indice)
	if S.Yendo then return end
	if not LP.Character then return end
	local HRP = LP.Character:FindFirstChild("HumanoidRootPart")
	if not HRP then return end
	local rancho = RANCHOS[indice]
	if not rancho then return end

	S.Yendo = true
	Estado.Text = "🥚 YENDO A: " .. rancho.nombre .. "..."

	for _, parte in pairs(LP.Character:GetDescendants()) do
		if parte:IsA("BasePart") then
			parte.CanCollide = false
		end
	end

	local destino = rancho.pos
	local intentos = 0

	while LP.Character and intentos < 500 do
		intentos = intentos + 1
		local actualHRP = LP.Character:FindFirstChild("HumanoidRootPart")
		if not actualHRP then break end

		local dist = (destino - actualHRP.Position).Magnitude
		if dist < 2 then
			actualHRP.CFrame = CFrame.new(destino)
			break
		end

		local dir = (destino - actualHRP.Position).Unit
		actualHRP.CFrame = CFrame.new(actualHRP.Position + dir * S.Velocidad)
		task.wait(0.01)
	end

	if LP.Character then
		local finalHRP = LP.Character:FindFirstChild("HumanoidRootPart")
		if finalHRP then finalHRP.CFrame = CFrame.new(destino) end
		task.wait(0.3)
		RestaurarColisiones()
	end

	S.Yendo = false
	Estado.Text = "✅ LLEGASTE A: " .. rancho.nombre
end

-- 🥚 DETECTAR HUEVO
local UltimoHuevo = 0
local function BuscarHuevo()
	if not S.FarmActivo then return end
	if S.Yendo then return end
	if not LP.Character then return end
	local HRP = LP.Character:FindFirstChild("HumanoidRootPart")
	if not HRP then return end

	for _, obj in pairs(Workspace:GetDescendants()) do
		if obj:IsA("BasePart") then
			local nom = obj.Name:lower()
			if nom:find("huevo") or nom:find("egg") then
				local dist = (obj.Position - HRP.Position).Magnitude
				if dist < 12 then
					local ahora = os.clock()
					if ahora - UltimoHuevo > 2 then
						UltimoHuevo = ahora
						task.wait(0.1)
						IrFarm(S.RanchoFarm)
					end
					return
				end
			end
		end
	end
end

local function LimpiarESP()
	for huevo, datos in pairs(ESPObjetos) do
		pcall(function()
			if datos.Cuadro then datos.Cuadro:Remove() end
			if datos.Texto then datos.Texto:Remove() end
		end)
		ESPObjetos[huevo] = nil
	end
end

local function CrearESP(huevo)
	if ESPObjetos[huevo] then return end
	local datos = {}
	pcall(function()
		if Drawing then
			datos.Cuadro = Drawing.new("Square")
			datos.Cuadro.Thickness = 2
			datos.Cuadro.Filled = false
			datos.Cuadro.Visible = false
			datos.Cuadro.Color = Color3.fromRGB(255, 200, 50)
			
			datos.Texto = Drawing.new("Text")
			datos.Texto.Size = 14
			datos.Texto.Center = true
			datos.Texto.Outline = true
			datos.Texto.Visible = false
			datos.Texto.Color = Color3.fromRGB(255, 255, 255)
		end
	end)
	if datos.Cuadro or datos.Texto then
		ESPObjetos[huevo] = datos
	end
end

local function ActualizarESP()
	if not S.ESP then
		for _, datos in pairs(ESPObjetos) do
			pcall(function()
				if datos.Cuadro then datos.Cuadro.Visible = false end
				if datos.Texto then datos.Texto.Visible = false end
			end)
		end
		return
	end
	if not LP.Character then return end
	local HRP = LP.Character:FindFirstChild("HumanoidRootPart")
	if not HRP then return end
	Camera = Workspace.CurrentCamera
	
	local huevosEncontrados = {}
	for _, obj in pairs(Workspace:GetDescendants()) do
		if obj:IsA("BasePart") then
			local nom = obj.Name:lower()
			if nom:find("huevo") or nom:find("egg") then
				huevosEncontrados[obj] = true
				CrearESP(obj)
				local datos = ESPObjetos[obj]
				if datos then
					local pos, enPantalla = Camera:WorldToViewportPoint(obj.Position)
					local distancia = (obj.Position - HRP.Position).Magnitude
					if enPantalla and distancia < 500 then
						local tam = math.max(20, 40 - distancia / 10)
						if datos.Cuadro then
							datos.Cuadro.Size = Vector2.new(tam, tam)
							datos.Cuadro.Position = Vector2.new(pos.X - tam/2, pos.Y - tam/2)
							datos.Cuadro.Visible = true
							if distancia < 30 then
								datos.Cuadro.Color = Color3.fromRGB(80, 255, 80)
							elseif distancia < 80 then
								datos.Cuadro.Color = Color3.fromRGB(255, 200, 50)
							else
								datos.Cuadro.Color = Color3.fromRGB(255, 80, 80)
							end
						end
						if datos.Texto then
							datos.Texto.Text = "🥚 " .. math.floor(distancia) .. " studs"
							datos.Texto.Position = Vector2.new(pos.X, pos.Y + tam/2 + 2)
							datos.Texto.Visible = true
						end
					else
						if datos.Cuadro then datos.Cuadro.Visible = false end
						if datos.Texto then datos.Texto.Visible = false end
					end
				end
			end
		end
	end
	
	for huevo, datos in pairs(ESPObjetos) do
		if not huevosEncontrados[huevo] or not huevo:IsDescendantOf(Workspace) then
			pcall(function()
				if datos.Cuadro then datos.Cuadro:Remove() end
				if datos.Texto then datos.Texto:Remove() end
			end)
			ESPObjetos[huevo] = nil
		end
	end
end
